int getStateID(ball, paddle) {
    int ballX = ball.Cx / 300; // 750/300 = 2
    int ballY = ball.Cy / 100; // 400/100 = 4
    int padY = paddle.y / 100; // 600/100 = 6
    int stateID = ballX + (ballY*4) + (padY*32);
    return stateID;
}

int chooseAction(int stateID) {
    float r = GetRandomValue(0, 100) / 100.0f;
    if (r < epsilon) {
        return GetRandomValue(0, 2); // EXPLORE
    } else {
        // EXPLOIT
        float maxQ = qTable[stateID][0];
        int bestAction = 0;
        for (int i = 1; i < 3; i++) {
            if (qTable[stateID][i] > maxQ) {
                maxQ = qTable[stateID][i]; bestAction = i;
            }
        }
        return bestAction;
    }
}

void updateQTable(int state, int act, int nextState, float reward) {
    float maxFutureQ = qTable[nextState][0];
    for (int i = 1; i < 3; i++) {
        if (qTable[nextState][i] > maxFutureQ)
            maxFutureQ = qTable[nextState][i];
    }
    float currentQ = qTable[state][act];
    float learnedValue = reward + (0.9f * maxFutureQ);
    qTable[state][act] = currentQ + 0.1f * (learnedValue - currentQ);
}
